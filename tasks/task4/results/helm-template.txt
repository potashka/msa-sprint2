apiVersion: apps/v1
kind: Deployment
metadata:
  name: booking-service
  labels:
    app: booking-service
    app.kubernetes.io/name: booking-service
    app.kubernetes.io/instance: booking-service
spec:
  replicas: 1
  selector:
    matchLabels:
      app: booking-service
  template:
    metadata:
      labels:
        app: booking-service
        app.kubernetes.io/name: booking-service
        app.kubernetes.io/instance: booking-service
    spec:
      containers:
        - name: booking-service
          image: "booking-service:latest"
          imagePullPolicy: Never
          ports:
            - name: http
              containerPort: 8080
          env:
            - name: ENABLE_FEATURE_X
              value: "false"
          resources:
            limits:
              cpu: 250m
              memory: 128Mi
            requests:
              cpu: 50m
              memory: 64Mi
          livenessProbe:
            httpGet:
              path: /ping
              port: http
            initialDelaySeconds: 5
            periodSeconds: 10
            timeoutSeconds: 2
            failureThreshold: 3
          readinessProbe:
            httpGet:
              path: /ready
              port: http
            initialDelaySeconds: 3
            periodSeconds: 5
            timeoutSeconds: 2
